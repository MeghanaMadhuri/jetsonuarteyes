# =============================================================================
# Jetson "Nina app" Python stack — Sirena UI + embedded tablet gateway + Wi‑Fi
# companion (Android). Install on the robot, not on the tablet.
#
#   python3 -m venv ~/nina-venv && source ~/nina-venv/bin/activate
#   pip install -U pip setuptools wheel
#   pip install -r requirements-app.txt
#   export PYTHONPATH=/path/to/Nvidia-jetson-platform
#
# Jetson / PyTorch: if `ultralytics` or `torch` wheels fail, install the NVIDIA
# wheel for your JetPack first, then re-run pip (see NVIDIA Jetson forums).
#
# Wi‑Fi reality: no software can guarantee zero packet loss or zero latency on
# a shared radio. Use 5 GHz when possible, place AP and robot/tablet with good
# RSSI, and keep firmware current. This file pulls deps that keep the link
# stable (HTTP stack, discovery, vision); tune AP QoS for robot traffic if needed.
#
# Recommended env for Android parity over Wi‑Fi (systemd drop-in or shell):
#   export NINA_ANDROID_GATEWAY_ALL=1
#   export NINA_LINK_MDNS=1
#   export NINA_LINK_LOG_UI_POLL_GETS=0
#   (optional shared secret) export NINA_ANDROID_HTTP_TOKEN=...
#
# Cross-check HTTP surface (run on Jetson after starting Sirena / gateway):
#   bash scripts/verify-nina-link-companion.sh
# =============================================================================
